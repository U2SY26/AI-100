## 🔬 검증 스크립트

### Python 코드 예시

```python
def verify_question3(menu_list):
    """
    문제 3번 검증 함수

    Args:
        menu_list: 1-2월 전체 메뉴명 리스트

    Returns:
        2회 이상 등장한 지역명 리스트
    """
    regions = ["안동", "전주", "베트남", "나가사키", "태국"]
    region_counts = {region: 0 for region in regions}

    # 각 메뉴명에서 지역명 검색
    for menu_name in menu_list:
        for region in regions:
            if region in menu_name:
                region_counts[region] += 1

    # 2회 이상만 필터링
    result = [
        region for region, count in region_counts.items()
        if count >= 2
    ]

    return sorted(result)  # 가나다순 정렬

# 사용 예시
all_menus = [
    "안동찜닭정식",
    "전주식콩나물국밥",
    "베트남쌀국수",
    "나가사키짬뽕",
    "태국식쌀국수",
    "데리야끼안동찜닭",
    "전주식비빔밥",
    "베트남쌀국수",  # 중복
    "나가사키짬뽕",   # 중복
]

answer = verify_question3(all_menus)
print("답:", answer)
# 출력: ['나가사키', '베트남', '안동', '전주']
```
